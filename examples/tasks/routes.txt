# vov route manifest — every endpoint this app declares, and who may reach it.
# Regenerate whenever a route declaration changes; review the diff.
#
#   METHOD  PATH  auth:<mode>  stack:<name>  [roles-any:a|b]  [perms-all:x,y]
#                 [min-tier:N]  [query:Type]  [body:Type]  [mcp-tool:name]
#
# roles-any  is satisfied by ANY one of the listed roles       (| reads as "or")
# perms-all  requires EVERY one of the listed permissions      (, reads as "and")
# min-tier   requires User.Tier() >= N; refused with 402, not 403
# query/body the Go type the input shape was declared from
# mcp-tool   the endpoint is callable by an AI assistant under this name

GET     /healthz                       auth:none  stack:bare
GET     /reports                       auth:required  stack:default  roles-any:member  min-tier:2  mcp-tool:get_reports
GET     /summary                       auth:required  stack:default
GET     /tasks                         auth:required  stack:default  query:listTasksQuery  mcp-tool:list_tasks
POST    /tasks                         auth:required  stack:json  perms-all:tasks.write  body:createTaskInput  mcp-tool:create_task
GET     /tasks/{id}                    auth:required  stack:default  mcp-tool:get_task
DELETE  /tasks/{id}                    auth:required  stack:default  roles-any:admin|owner  perms-all:tasks.write  mcp-tool:delete_task
POST    /webhook                       auth:none  stack:webhook
